Apps Script কোড যুক্ত করুন

Google Sheet-এর ভেতর থেকে Apps Script এডিটর খুলুন:

  1. শীটের উপরের মেন্যু থেকে Extensions > Apps Script-এ ক্লিক করুন।
  2. এডিটরে পূর্বে থাকা function myFunction() { } লিখা সম্পূর্ণ মুছে ফেলুন।
  3. নিচের বাটনে ক্লিক করে পুরো কোডটি কপি করে সেখানে পেস্ট করুন এবং Ctrl + S চেপে সেভ করুন:

 * 
 * SETUP INSTRUCTIONS:
 *   1. Create a new Google Spreadsheet and name it "Pexora Backup".
 *   2. Go to: Extensions > Apps Script.

 *   4. Click "Deploy" > "New deployment".
 *   5. Select type: "Web app".
 *   6. Set "Execute as": "Me".
 *   7. Set "Who has access": "Anyone" (Required for Pexora to communicate).
 *   8. Click "Deploy", authorize access, and copy the "Web app URL".
 *   9. In Pexora, click "Backup" button and paste your Web app URL. Done!
 * ==============================================================================
 */

var TAB_PROMPTS = "Prompts";
var TAB_CATEGORIES = "Categories";
var TAB_RAW_BACKUP = "_raw_snapshot";

function doGet(e) {
  return handleRequest(e);
}

function doPost(e) {
  return handleRequest(e);
}

function handleRequest(e) {
  try {
    var params = (e && e.parameter) ? e.parameter : {};
    var postData = {};
    if (e && e.postData && e.postData.contents) {
      try {
        postData = JSON.parse(e.postData.contents);
      } catch (err) {
        // Fallback for url-encoded form post
      }
    }

    var data = Object.assign({}, params, postData);
    var action = (data.action || '').trim().toLowerCase();

    // Browser visit without action -> Render Web Dashboard
    if (!action || action === 'dashboard') {
      return renderDashboard();
    }

    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (action === 'backup' || action === 'sync_push') {
      return handleBackup(ss, data);
    } else if (action === 'restore' || action === 'get_backup' || action === 'sync_pull') {
      return handleRestore(ss, data);
    } else if (action === 'stats' || action === 'ping') {
      return handleStats(ss);
    } else {
      return jsonResponse({
        success: false,
        code: 'UNKNOWN_ACTION',
        message: 'Action not supported. Available actions: backup, restore, stats, ping.'
      });
    }
  } catch (err) {
    return jsonResponse({
      success: false,
      code: 'SERVER_ERROR',
      message: 'Server error: ' + (err.message || String(err))
    });
  }
}

/**
 * Handle incoming backup payload from Pexora
 */
function handleBackup(ss, data) {
  var now = new Date();
  var timestampIso = now.toISOString();

  var promptsData = (data.prompts && typeof data.prompts === 'object') ? data.prompts : {};
  var folderMetadata = (data.folder_metadata && typeof data.folder_metadata === 'object') ? data.folder_metadata : {};
  var settings = (data.settings && typeof data.settings === 'object') ? data.settings : { view: 'grid' };

  var totalPrompts = countTotalPrompts(promptsData);

  // CLOUD PROTECTION: If incoming prompts are 0, but sheet already has prompts, reject empty push!
  if (totalPrompts === 0) {
    var existingRaw = readRawBackup(ss);
    if (existingRaw && countTotalPrompts(existingRaw.prompts) > 0) {
      return jsonResponse({
        success: false,
        code: 'CLOUD_PROTECTED',
        message: 'Empty backup rejected: Google Sheet already has ' + countTotalPrompts(existingRaw.prompts) + ' prompts. Sheet protected.'
      });
    }
  }

  // 1. Update Prompts Sheet
  var totalPromptsUpdated = updatePromptsSheet(ss, promptsData, folderMetadata);

  // 2. Update Categories Sheet
  var totalCategories = updateCategoriesSheet(ss, folderMetadata);

  // 3. Save Full Lossless Raw Snapshot
  var rawPayload = {
    schema_version: data.schema_version || 3,
    timestamp: timestampIso,
    prompts: promptsData,
    folder_metadata: folderMetadata,
    settings: settings
  };
  saveRawBackup(ss, rawPayload);

  return jsonResponse({
    success: true,
    code: 'BACKUP_SUCCESS',
    message: 'Pexora backup successfully saved to Google Sheets!',
    timestamp: timestampIso,
    counts: {
      prompts: totalPromptsUpdated,
      categories: totalCategories
    }
  });
}

/**
 * Handle restore request from Pexora
 */
function handleRestore(ss, data) {
  // 1. Try reading lossless raw snapshot first
  var rawData = readRawBackup(ss);
  if (rawData && rawData.prompts && Object.keys(rawData.prompts).length > 0) {
    var promptCount = countTotalPrompts(rawData.prompts);
    return jsonResponse({
      success: true,
      code: 'RESTORE_SUCCESS',
      source: 'raw_snapshot',
      backup: rawData,
      totalPrompts: promptCount
    });
  }

  // 2. Fallback: Reconstruct from formatted table sheets
  var reconstructed = reconstructFromSheets(ss);
  var count = countTotalPrompts(reconstructed.prompts);

  return jsonResponse({
    success: true,
    code: 'RESTORE_SUCCESS',
    source: 'table_reconstruction',
    backup: reconstructed,
    totalPrompts: count
  });
}

/**
 * Handle stats / healthcheck ping
 */
function handleStats(ss) {
  var promptSheet = ss.getSheetByName(TAB_PROMPTS);
  var promptCount = promptSheet ? Math.max(0, promptSheet.getLastRow() - 1) : 0;
  
  var catSheet = ss.getSheetByName(TAB_CATEGORIES);
  var catCount = catSheet ? Math.max(0, catSheet.getLastRow() - 1) : 0;

  var rawData = readRawBackup(ss);
  if (rawData && rawData.prompts) {
    var snapshotCount = countTotalPrompts(rawData.prompts);
    if (snapshotCount > 0) promptCount = snapshotCount;
  }

  return jsonResponse({
    success: true,
    sheetName: ss.getName(),
    totalPrompts: promptCount,
    totalCategories: catCount,
    lastBackupAt: rawData ? rawData.timestamp : null
  });
}

function getOrCreateSheet(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }
  return sheet;
}

function updatePromptsSheet(ss, promptsData, metadata) {
  var sheet = getOrCreateSheet(ss, TAB_PROMPTS);
  sheet.clearContents();

  var headers = [
    'Category', 'Subcategory', 'Title', 'Keywords', 'Prompt Content',
    'Prompt Note', 'Favorite', 'Variable Names', 'Variable Options',
    'Order', 'Updated At', 'Prompt ID'
  ];

  var rows = [headers];
  var total = 0;

  for (var folder in promptsData) {
    if (!promptsData.hasOwnProperty(folder)) continue;
    var list = promptsData[folder];
    if (!Array.isArray(list)) continue;

    for (var i = 0; i < list.length; i++) {
      var p = list[i];
      if (!p) continue;
      total++;

      var varNamesStr = '';
      if (p.variable_display_names && typeof p.variable_display_names === 'object') {
        var pairs = [];
        for (var k in p.variable_display_names) {
          if (p.variable_display_names.hasOwnProperty(k)) {
            pairs.push(k + ': ' + p.variable_display_names[k]);
          }
        }
        varNamesStr = pairs.join(' | ');
      }

      var varOptsStr = '';
      if (p.variable_options && typeof p.variable_options === 'object') {
        var optPairs = [];
        for (var ok in p.variable_options) {
          if (p.variable_options.hasOwnProperty(ok)) {
            var opts = p.variable_options[ok];
            optPairs.push(ok + ': [' + (Array.isArray(opts) ? opts.join(', ') : String(opts)) + ']');
          }
        }
        varOptsStr = optPairs.join(' | ');
      }

      rows.push([
        p.category || folder || 'General',
        p.subcategory || '',
        p.title || 'Untitled Prompt',
        p.keywords || '',
        p.prompt || '',
        p.note || '',
        p.favorite ? 'Yes' : 'No',
        varNamesStr,
        varOptsStr,
        p.order || 0,
        p.updated_at || '',
        p.id || ''
      ]);
    }
  }

  if (rows.length > 0) {
    sheet.getRange(1, 1, rows.length, headers.length).setValues(rows);
    formatHeaderRow(sheet, headers.length, '#4f46e5');
    sheet.setFrozenRows(1);
    sheet.setFrozenColumns(1);
  }

  return total;
}

function updateCategoriesSheet(ss, metadata) {
  var sheet = getOrCreateSheet(ss, TAB_CATEGORIES);
  sheet.clearContents();

  var headers = ['Category / Folder Name', 'Parent Category', 'Color Hex', 'Category ID'];
  var rows = [headers];
  var count = 0;

  for (var name in metadata) {
    if (!metadata.hasOwnProperty(name)) continue;
    var meta = metadata[name] || {};
    count++;
    rows.push([
      name,
      meta.parent || '',
      meta.color || '#2979FF',
      meta.id || ''
    ]);
  }

  if (rows.length > 0) {
    sheet.getRange(1, 1, rows.length, headers.length).setValues(rows);
    formatHeaderRow(sheet, headers.length, '#059669');
    sheet.setFrozenRows(1);
    sheet.setFrozenColumns(1);
  }

  return count;
}

function formatHeaderRow(sheet, colCount, bgHex) {
  try {
    var range = sheet.getRange(1, 1, 1, colCount);
    range.setBackground(bgHex || '#4f46e5');
    range.setFontColor('#ffffff');
    range.setFontWeight('bold');
    range.setFontFamily('Segoe UI');
    range.setFontSize(10);
  } catch (_) {}
}

function saveRawBackup(ss, payload) {
  var sheet = getOrCreateSheet(ss, TAB_RAW_BACKUP);
  sheet.clearContents();
  sheet.getRange(1, 1).setValue('Pexora Raw Backup Snapshot');
  sheet.getRange(1, 2).setValue(new Date().toISOString());

  // Store JSON split into 40KB safe cell chunks
  var jsonStr = JSON.stringify(payload);
  var chunkSize = 40000;
  var chunks = [];
  for (var i = 0; i < jsonStr.length; i += chunkSize) {
    chunks.push([jsonStr.substring(i, i + chunkSize)]);
  }
  if (chunks.length > 0) {
    sheet.getRange(2, 1, chunks.length, 1).setValues(chunks);
  }
  sheet.hideSheet();
}

function readRawBackup(ss) {
  try {
    var sheet = ss.getSheetByName(TAB_RAW_BACKUP);
    if (!sheet) return null;
    var lastRow = sheet.getLastRow();
    if (lastRow < 2) return null;
    var values = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
    var full = '';
    for (var i = 0; i < values.length; i++) {
      full += String(values[i][0] || '');
    }
    return JSON.parse(full);
  } catch (err) {
    return null;
  }
}

function reconstructFromSheets(ss) {
  var promptSheet = ss.getSheetByName(TAB_PROMPTS);
  var prompts = {};
  var metadata = {};

  // Read Categories
  var catSheet = ss.getSheetByName(TAB_CATEGORIES);
  if (catSheet && catSheet.getLastRow() > 1) {
    var catData = catSheet.getDataRange().getValues();
    for (var c = 1; c < catData.length; c++) {
      var cName = String(catData[c][0] || '').trim();
      if (!cName) continue;
      metadata[cName] = {
        parent: String(catData[c][1] || ''),
        color: String(catData[c][2] || '#2979FF'),
        id: String(catData[c][3] || '')
      };
    }
  }

  // Read Prompts
  if (promptSheet && promptSheet.getLastRow() > 1) {
    var data = promptSheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var cat = String(row[0] || 'General').trim() || 'General';
      var sub = String(row[1] || '').trim();
      var folder = sub || cat;

      prompts[folder] = prompts[folder] || [];
      prompts[folder].push({
        id: String(row[11] || ('pex-' + i)),
        category: cat,
        subcategory: sub,
        folder: folder,
        title: String(row[2] || 'Untitled Prompt'),
        keywords: String(row[3] || ''),
        prompt: String(row[4] || ''),
        prompt_html: '<p>' + String(row[4] || '').replace(/\n/g, '<br>') + '</p>',
        note: String(row[5] || ''),
        favorite: String(row[6] || '').toLowerCase() === 'yes',
        variable_display_names: {},
        variable_options: {},
        variable_notes: {},
        order: Number(row[9]) || 0,
        updated_at: String(row[10] || new Date().toISOString())
      });

      metadata[cat] = metadata[cat] || { id: '', color: '#2979FF', parent: '' };
      if (sub) {
        metadata[sub] = metadata[sub] || { id: '', color: metadata[cat].color, parent: cat };
      }
    }
  }

  return {
    schema_version: 3,
    timestamp: new Date().toISOString(),
    prompts: prompts,
    folder_metadata: metadata,
    settings: { view: 'grid' }
  };
}

function countTotalPrompts(promptsObj) {
  var count = 0;
  for (var k in promptsObj) {
    if (promptsObj.hasOwnProperty(k) && Array.isArray(promptsObj[k])) {
      count += promptsObj[k].length;
    }
  }
  return count;
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

Web App হিসেবে ডিপ্লয় (Deploy) করুন

Apps Script এডিটরের উপরের ডানদিকের নীল Deploy বাটনে ক্লিক করুন:

  1. Deploy > New deployment এ ক্লিক করুন।
  2. বামপাশের গিয়ার (Gear) আইকনে ক্লিক করে Web app সিলেক্ট করুন।
  3. সেটিংসগুলো এভাবে পূরণ করুন:
    • Description: Pexora Backup
    • Execute as: Me (আপনার ইমেইল)
    • Who has access: Anyone (অবশ্যই Anyone সিলেক্ট করতে হবে)
  4. Deploy বাটনে চাপ দিন এবং প্রম্পট আসলে গুগলের পারমিশন দিন (Authorize Access > Advanced > Go to ... (unsafe) > Allow)।
  5. ডিপ্লয় সম্পন্ন হলে দেওয়া Web app URL টি কপি করে নিন। (এটি দেখতে https://script.google.com/macros/s/.../exec এর মতো হবে)।
⚠️ গুরুত্বপূর্ণ: "Who has access" অপশনে অবশ্যই Anyone সিলেক্ট করবেন। তা না হলে Pexora সফটওয়্যার ব্যাকআপ ডাটা পাঠাতে পারবে না।

Pexora সফটওয়্যারে কানেক্ট করুন

এখন Pexora সফটওয়্যারটি ওপেন করুন:

  1. উপরের ডানপাশের Backup বাটনে ক্লিক করুন।
  2. ইনপুট বক্সে আপনার কপি করা Web app URL টি পেস্ট করুন।
  3. Connect & Test বাটনে চাপ দিন।
  4. সবুজ রঙে Connected! Sheet: "Pexora Backup" দেখাবে।
  5. এবার Backup to Google Sheet এ ক্লিক করলেই আপনার সমস্ত প্রম্পট ক্লাউডে সুরক্ষিত হয়ে যাবে!
